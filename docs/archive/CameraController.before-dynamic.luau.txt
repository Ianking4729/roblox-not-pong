-- CameraController: third-person camera above and behind the local player, aimed at the table.
-- Visual only; it reads Config and the local character, and never touches gameplay state.
--
-- Placement (world axes, because the table is axis-aligned and the player always faces -Z):
--   position = player root + (ShoulderOffset, -, DistanceBehind), at Height above the floor
--   aim point = table surface centre (+ LookAhead toward the opponent), blended sideways by LateralFollow
-- ShoulderOffset shifts the camera off the player's centre line so the avatar sits to one side of the screen
-- instead of covering the near end of the table.

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")

local Config = require(ReplicatedStorage:WaitForChild("Shared"):WaitForChild("Config"))

local BIND_NAME = "NotPongCamera"

local CameraController = {}
CameraController.__index = CameraController

function CameraController.new(player: Player)
	return setmetatable({
		player = player,
		position = nil :: Vector3?,
		lookAt = nil :: Vector3?,
		fadeParts = {} :: { [BasePart]: number }, -- part -> local transparency
		fadeCharacter = nil :: Model?,
		fadeConn = nil :: RBXScriptConnection?,
	}, CameraController)
end

-- Where the camera wants to be and look, given the player's root position. Pure; used by tests too.
function CameraController.compute(rootPosition: Vector3): (Vector3, Vector3)
	local C, L = Config.Camera, Config.Layout
	local desiredPosition = Vector3.new(rootPosition.X + C.ShoulderOffset, L.FloorY + C.Height, rootPosition.Z + C.DistanceBehind)
	local aimX = L.TableSurfaceCenter.X + (rootPosition.X - L.TableSurfaceCenter.X) * C.LateralFollow
	local desiredLookAt = Vector3.new(
		aimX,
		L.SurfaceY + C.LookAtHeightOffset,
		L.TableSurfaceCenter.Z - C.LookAheadStuds
	)
	return desiredPosition, desiredLookAt
end

function CameraController:_refreshFade(character: Model?)
	if self.fadeCharacter == character then
		return
	end
	if self.fadeConn then
		self.fadeConn:Disconnect()
		self.fadeConn = nil
	end
	self.fadeParts = {}
	self.fadeCharacter = character
	if not character then
		return
	end
	-- The paddle stays opaque: it is a gameplay object the player needs to see. The head (and anything worn on
	-- it) fades less than the body so the silhouette stays readable.
	local C = Config.Camera
	local function fadeFor(d: Instance): number?
		if not d:IsA("BasePart") or d:FindFirstAncestor("Paddle") then
			return nil
		end
		if d.Name == "Head" then
			return C.HeadTransparency
		end
		local weld = d:FindFirstChild("AccessoryWeld")
		if weld and weld:IsA("Weld") and weld.Part1 and weld.Part1.Name == "Head" then
			return C.HeadTransparency
		end
		return C.BodyTransparency
	end
	local function track(d: Instance)
		local t = fadeFor(d)
		if t then
			self.fadeParts[d :: BasePart] = t
		end
	end
	for _, d in character:GetDescendants() do
		track(d)
	end
	self.fadeConn = character.DescendantAdded:Connect(function(d)
		task.defer(track, d) -- accessory welds appear a moment after the handle
	end)
end

function CameraController:_update(dt: number)
	local character = self.player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart") :: BasePart?
	if not root then
		return
	end
	local camera = workspace.CurrentCamera
	if not camera then
		return
	end

	local C = Config.Camera
	local desiredPosition, desiredLookAt = CameraController.compute(root.Position)
	if not self.position or not self.lookAt then
		self.position, self.lookAt = desiredPosition, desiredLookAt
	else
		local alpha = 1 - math.exp(-C.Smoothing * dt)
		self.position = self.position:Lerp(desiredPosition, alpha)
		self.lookAt = self.lookAt:Lerp(desiredLookAt, alpha)
	end

	camera.CameraType = Enum.CameraType.Scriptable
	camera.FieldOfView = C.FieldOfView
	camera.CFrame = CFrame.lookAt(self.position, self.lookAt)

	self:_refreshFade(character)
	for part, t in self.fadeParts do
		part.LocalTransparencyModifier = t
	end
end

function CameraController:start()
	RunService:BindToRenderStep(BIND_NAME, Enum.RenderPriority.Camera.Value + 1, function(dt)
		self:_update(dt)
	end)
end

function CameraController:stop()
	RunService:UnbindFromRenderStep(BIND_NAME)
	self:_refreshFade(nil)
end

return CameraController
